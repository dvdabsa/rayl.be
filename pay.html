04 — Integrate

Live in
minutes.

Drop a script tag, mount a button, take payments. That's the entire flow.

checkout.html 
<!-- 1. Drop in Rayl Pay -->
<script src="https://js.raylpay.io/v1"></script>

<!-- 2. Mount a checkout button -->
<div id="pay"></div>

<script>
  const rayl = new RaylPay({
    publicKey: "pk_live_...",
    methods: ["card", "apple_pay", "google_pay"],
    amount: 4900,        // in cents
    currency: "EUR",
    onSuccess: (tx) => console.log("paid", tx.id),
  });

  rayl.mount("#pay");
</script>
